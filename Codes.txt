<script>
const scenes = Array.from(document.querySelectorAll(".comic-scene"));
const fullTexts = scenes.map(scene => scene.dataset.text.trim());
const textElements = scenes.map(scene => scene.querySelector(".comic-text"));
const pictures = scenes.map(scene => scene.querySelector(".comic-picture"));

const audioFile = document.getElementById("audioFile");
const volumeSlider = document.getElementById("myRange");

let currentScene = 0;
let textIndex = 0;
let typingInterval = null;
let typing = false;
let animationFinished = false;
let transitioning = false;
let hasStarted = false;

const typingSpeed = 10;
const transitionDuration = 3000;

// Set the starting volume to match the slider.
audioFile.volume = volumeSlider.value / 100;

// Change music volume while moving the slider.
volumeSlider.addEventListener("input", () => {
  audioFile.volume = volumeSlider.value / 100;
});

function startTyping() {
  const textBox = textElements[currentScene];

  textBox.style.display = "block";
  textBox.textContent = "";

  clearInterval(typingInterval);

  textIndex = 0;
  typing = true;

  typingInterval = setInterval(() => {
    const fullText = fullTexts[currentScene];

    if (textIndex < fullText.length) {
      textBox.textContent += fullText.charAt(textIndex);
      textIndex++;
    } else {
      clearInterval(typingInterval);
      typingInterval = null;
      typing = false;
    }
  }, typingSpeed);
}

function setFinalPicturePosition(picture) {
  picture.style.animation = "none";
  picture.style.transform = "translate(-350px, -300px)";
}

function showScene(index) {
  const scene = scenes[index];
  const picture = pictures[index];

  scene.classList.add("active");

  picture.style.animation = "";
  picture.style.transform = "";

  animationFinished = false;  // Reset this here
  textIndex = 0;  // Add this line

  picture.addEventListener("animationend", function onEnd() {
    picture.removeEventListener("animationend", onEnd);

    if (currentScene === index) {
      animationFinished = true;

      if (!transitioning) {
        startTyping();
      }
    }
  });
}

function nextScene() {
  if (
    transitioning ||
    currentScene >= scenes.length - 1
  ) {
    return;
  }

  transitioning = true;

  clearInterval(typingInterval);
  typingInterval = null;
  typing = false;

  const oldScene = scenes[currentScene];
  const nextIndex = currentScene + 1;
  const newScene = scenes[nextIndex];

  currentScene = nextIndex;

  newScene.classList.add("active", "scene-fade-in");
  oldScene.classList.add("scene-fade-out");

  showScene(currentScene);

  setTimeout(() => {
    oldScene.classList.remove("active", "scene-fade-out");
    newScene.classList.remove("scene-fade-in");
    
    // Hide the old text box
    textElements[currentScene - 1].style.display = "none";
    textElements[currentScene - 1].textContent = "";

    transitioning = false;

    if (animationFinished) {
      startTyping();
    }
  }, transitionDuration);
}

document.addEventListener("keydown", event => {
  if (
    event.key.toLowerCase() !== "e" ||
    event.repeat
  ) {
    return;
  }

  // First E press starts the music and starts the first picture.
  if (!hasStarted) {
    hasStarted = true;

    audioFile.currentTime = 0;

    audioFile.play().catch(error => {
      console.error("The audio could not start:", error);
    });

    showScene(0);
    return;
  }

  if (transitioning) {
    return;
  }

  // If the picture is still moving, finish its animation.
  if (!animationFinished) {
    setFinalPicturePosition(pictures[currentScene]);

    animationFinished = true;
    startTyping();

    return;
  }

  // If the text is typing, immediately show all of it.
  if (typing) {
    clearInterval(typingInterval);
    typingInterval = null;

    textElements[currentScene].textContent =
      fullTexts[currentScene];

    typing = false;

    return;
  }

  // Otherwise, move to the next picture.
  nextScene();
});

let typing = false;
let animationFinished = false;
let transitioning = false;
let hasStarted = false;

const typingSpeed = 10;
const transitionDuration = 800;
const pictureAnimations = ["nothing", "slide-right", "slide-down", "slide-left"];

function startTyping() {

  textBox.style.display = "block";
  textBox.textContent = "";
  clearInterval(typingInterval);
  textIndex = 0;

  typingInterval = setInterval(() => {

    if (textIndex < fullText.length) {
      textBox.textContent += fullText.charAt(textIndex);

function restartPictureAnimation(index) {
  const picture = pictures[index];
  const animationClass = pictureAnimations[index] || "slide-right";

  picture.classList.remove(...pictureAnimations);
  void picture.offsetWidth; // force restart
  picture.classList.add(animationClass);
  picture.style.animation = "";
  picture.style.transform = "";
}

function setFinalPicturePosition(picture) {
  picture.classList.remove(...pictureAnimations);
  picture.style.animation = "none";
  
  // Set the final transform based on which scene we're on
  const finalPositions = [
    "transform: translate(0, 0)",
    "translate(-350px, -300px)",      // Scene 1 (slide-right) - start position is the end
    "translate(-350px, -10px)", // Scene 2 (slide-down) - top position
    "translate(-350px, -300px)"  // Scene 3 (slide-left) - left position
  ];
  
  picture.style.transform = finalPositions[currentScene];
}

function showScene(index) {
  const scene = scenes[index];
  const picture = pictures[index];

  scene.classList.add("active");
  restartPictureAnimation(index);

  animationFinished = false;

  picture.onanimationend = function () {
    if (currentScene === index) {
      animationFinished = true;

      if (!transitioning) {
        startTyping();
      }
    }
  };
}

function nextScene() {
  if (transitioning || currentScene >= scenes.length - 1) return;

  transitioning = true;

  clearInterval(typingInterval);
  typingInterval = null;
